function cetakTransaksi(t) { if (!t) { alert("Data transaksi tidak ditemukan."); return; } const rupiah = angka => "Rp " + (Number(angka) || 0).toLocaleString("id-ID"); const aman = teks => String(teks ?? "-") .replace(/[&<>"']/g, karakter => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[karakter]); const sisa = Math.max( 0, (Number(t.total) || 0) - (Number(t.dp) || 0) ); const rincian = [ ["PKB", t.pkb], ["Opsen PKB", t.opsenPkb], ["SWDKLLJ", t.swdkllj], ["Denda PKB", t.dendaPkb], ["Denda SWDKLLJ", t.dendaSwdkllj], ["BBNKB", t.bbnkb], ["Opsen BBNKB", t.opsenBbnkb], ["PNBP", t.pnbp], ["Biaya resmi lainnya", t.biayaResmiLainnya], ["Biaya resmi", t.biayaResmi], ["Jasa biro", t.biayaJasa ?? t.jasa], ["TOTAL TAGIHAN", t.total], ["DP / Uang muka", t.dp], ["SISA PEMBAYARAN", sisa] ]; const barisRincian = rincian.map(([label, nilai]) => ` ${aman(label)} ${rupiah(nilai)} `).join(""); const tanggal = t.tanggal ? new Date(t.tanggal).toLocaleString("id-ID") : new Date().toLocaleString("id-ID"); const halaman = ` Bukti Transaksi ASKARA ZAYYAN

ASKARA ZAYYAN

Biro Jasa Kendaraan Bermotor

BUKTI TRANSAKSI

Tanggal: ${aman(tanggal)}
Layanan: ${aman(t.layanan)}
Jenis kendaraan: ${aman(t.jenisKendaraan)}
Nama pengurus: ${aman(t.namaPengurus)}
Nama STNK: ${aman(t.namaStnk)}
Nomor polisi: ${aman(t.nopol)}
Merek kendaraan: ${aman(t.merek)}
Wilayah STNK: ${aman(t.wilayah)}
${barisRincian}
Rincian biayaJumlah

Status: ${aman(t.statusBerkas || t.status || "PROSES")}

Catatan: ${aman(t.catatan || "-")}

Terima kasih telah menggunakan jasa kami.
`; const jendela = window.open("", "_blank"); if (!jendela) { alert("Jendela cetak diblokir. Izinkan pop-up untuk situs ini, lalu coba lagi."); return; } jendela.document.open(); jendela.document.write(halaman); jendela.document.close(); }
